Player {turn} wins!" Element("msg").write(result) speak(result) active = False elif "" not in board: msg = "It's a draw!" Element("msg").write(msg) speak(msg) active = False else: turn = "O" if turn == "X" else "X" Element("msg").write(f"Player {turn}'s Turn") def check_win(): wins = [(0,1,2), (3,4,5), (6,7,8), (0,3,6), (1,4,7), (2,5,8), (0,4,8), (2,4,6)] for w in wins: if board[w[0]] == board[w[1]] == board[w[2]] != "": return True return False def restart(event): global board, turn, active board = [""] * 9 turn = "X" active = True Element("msg").write("New Game! Player X's Turn") speak("Game Restarted") # This clears the visual boxes for i in range(9): Element(str(i)).write("")
